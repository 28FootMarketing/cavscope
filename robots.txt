# /onboarding and /sitrep are behind Supabase Auth and return the signed-out
# state to a crawler, so they are excluded: indexing them would put an empty
# sign-in gate in search results under a title that promises a report.
# /sitrep/sample is the deliberate exception -- it is the static, fictional
# reference SITREP, and it is meant to be findable. Longest-match wins, so the
# Allow below beats the Disallow above it.

Disallow: /sitrep
Allow: /sitrep/sample
