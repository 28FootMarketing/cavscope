# cavscope.28footsystems.com -- one host serving both the public marketing
# site and the signed-in application. Merges what robots.txt and
# robots-app.txt do separately for muster.partners / app.muster.partners.
#
# /onboarding, /sitrep (except the sample), /app, /admin and /signin are all
# sign-in gates or tenant-scoped pages that return the signed-out state to a
# crawler; indexing them would put an empty gate in search results under a
# title that promises a report or a workspace. Longest-match wins, so the
# Allow below beats the Disallow above it.

User-agent: *
Disallow: /onboarding
Disallow: /sitrep
Disallow: /app
Disallow: /admin
Disallow: /signin
Allow: /sitrep/sample
